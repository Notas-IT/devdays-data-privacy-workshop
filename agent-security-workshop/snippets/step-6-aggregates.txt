import json, os, sys
from decimal import Decimal
from urllib.request import urlopen
from urllib.error import HTTPError

base = os.environ["GATEWAY_URL"]
with urlopen(base + "/companies/1/salary-summary", timeout=5) as response:
    summary = json.load(response)
if (set(summary) != {"company_id", "employee_count", "average_salary_eur"}
        or summary["company_id"] != 1 or summary["employee_count"] != 3
        or Decimal(str(summary["average_salary_eur"])) != Decimal("4000")):
    sys.exit("FAIL: unexpected company summary")
print("PASS: company 1 average is EUR 4000 across 3 employees")

try:
    with urlopen(base + "/companies/1/salaries", timeout=5):
        sys.exit("FAIL: individual salaries are accessible")
except HTTPError as error:
    if error.code != 403:
        raise
print("PASS: individual salary requests are blocked")

with urlopen(base + "/projects", timeout=5) as response:
    if not json.load(response)["items"]:
        sys.exit("FAIL: public projects are unavailable")
print("PASS: public projects remain readable")
'
